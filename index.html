function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  
  // Crear encabezados automáticamente si la hoja está vacía
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      "FECHA Y HORA",
      "NOMBRES Y APELLIDOS",
      "CURSO",
      "PARALELO",
      "CURSO COMPLETO",
      "PUNTAJE (SOBRE 45)",
      "ACIERTOS (DE 15)",
      "PREGUNTAS FALLADAS",
      "DETALLE DE RESPUESTAS"
    ]);
  }

  var data = JSON.parse(e.postData.contents);

  sheet.appendRow([
    data.fechaHora || new Date().toLocaleString("es-BO"),
    data.nombre || "",
    data.curso || "",
    data.paralelo || "",
    data.cursoCompleto || "",
    Number(data.puntaje) || 0,
    data.aciertosTexto || "",
    data.preguntasFalladas || "",
    data.detalleRespuestas || ""
  ]);

  return ContentService
    .createTextOutput(JSON.stringify({ status: "ok" }))
    .setMimeType(ContentService.MimeType.JSON);
}
